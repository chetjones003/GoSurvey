# GitHub issue #402 / REQ-329 increment 3 — SCALE is uniform about the UCS-resolved base point.
#
# The base point carries its picked elevation, and the elevations scale about it. Since issue #1577
# that holds in plan view too: SCALE is uniform in X, Y and Z, as AutoCAD's is.

# --- Regression: World UCS, SCALE 2 about the origin --------------------------------------------
NEW
CMD LINE
CMD 10,0
CMD 20,0
ESC
BOX 5 -5 25 5
EXPECT SELECTED 1
CMD SCALE
CMD 0,0
CMD 2
ESC
EXPECT LINEXYZ 0 20 0 0 40 0 0
CHECK ALL

# --- World UCS, a line WITH elevation: Z scales about the base too (AutoCAD parity, #1577) -------
NEW
CMD LINE
CMD 10,0
CMD 20,0
ESC
BOX 5 -5 25 5
CMD MOVE
CMD 0,0
CMD 0,0,5
ESC
EXPECT LINEXYZ 0 10 0 5 20 0 5
BOX 5 -5 25 5
CMD SCALE
CMD 0,0
CMD 2
ESC
# X doubles about the origin, and so does the elevation: AutoCAD's SCALE is uniform in X, Y and Z
# (a line at z 3 scaled 2 from the origin ends at z 6 - tools/autocad-transform-oracle/).
EXPECT LINEXYZ 0 20 0 10 40 0 10
CHECK ALL

# --- Tilted UCS (X 90): a line along the UCS Y axis (world Z) scales its elevation ---------------
# UCS X 90: basis X=(1,0,0), Y=(0,0,1), Z=(0,-1,0). Line UCS (0,0)-(0,10) = world (0,0,0)-(0,0,10).
# SCALE 2 about UCS (0,0) = world origin. World X/Y are unchanged (both 0); the Z scales about 0:
# (0,0,10) -> (0,0,20). Before this change the far endpoint would stay at (0,0,10).
NEW
CMD UCS
CMD X
CMD 90
CMD LINE
CLICKUCS 0 0
CLICKUCS 0 10
ESC
BOX -5 -5 5 5
EXPECT SELECTED 1
CMD SCALE
CLICKUCS 0 0
CMD 2
ESC
CMD UCS
CMD W
EXPECT LINEXYZ 0 0 0 0 0 0 20
CHECK ALL

# --- Tilted UCS: a circle off the base plane — centre elevation and radius both scale -----------
# Circle UCS (0,10) = world (0,0,10), r 2. SCALE 2 about world origin: centre (0,0,10) -> (0,0,20),
# r 2 -> 4. Normal (0,-1,0) is unchanged (a uniform scale does not tip the plane).
NEW
CMD UCS
CMD X
CMD 90
CMD CIRCLE
CLICKUCS 0 10
CMD 2
ESC
BOX -5 -5 5 5
EXPECT SELECTED 1
CMD SCALE
CLICKUCS 0 0
CMD 2
ESC
CMD UCS
CMD W
EXPECT CIRCLES 1
EXPECT CIRCLEXYZ 0 0 0 20 4 0 -1 0
CHECK ALL

# --- Tilted UCS: SCALE about a NON-ZERO base elevation -----------------------------------------
# Line UCS (0,10)-(0,20) = world (0,0,10)-(0,0,20). Base UCS (0,5) = world (0,0,5). SCALE 3:
# Z: 10 -> 5 + 3*(10-5) = 20; 20 -> 5 + 3*(20-5) = 50.
NEW
CMD UCS
CMD X
CMD 90
CMD LINE
CLICKUCS 0 10
CLICKUCS 0 20
ESC
BOX -5 -5 5 5
CMD SCALE
CLICKUCS 0 5
CMD 3
ESC
CMD UCS
CMD W
EXPECT LINEXYZ 0 0 0 20 0 0 50
CHECK ALL

# --- UCS rotated 30 about Z (Z still world Z) scales elevations, like plan view -----------------
NEW
CMD LINE
CMD 10,0
CMD 20,0
ESC

CMD 0,0
CMD 0,0,5

CMD UCS

CMD UCS
CMD W
# The typed base `0,0` still lands at the world origin under the Z-30 UCS, so this doubles about the
# origin in X, Y and Z: elevation 5 -> 10, the same as plan view (AutoCAD parity, issue #1577).
EXPECT LINEXYZ 0 20 0 10 40 0 10
CHECK ALL
