# GitHub #150 (3D Phase 7) — PADSOLID asks for what it needs, the way every other command does.
#
# The first version took everything on one line and, typed bare, printed a usage line and ended. From
# the app that reads as the command doing nothing: you type PADSOLID, press Enter, and the command is
# already over before you can give it an elevation. Reported from the app, 2026-10-02.
#
# It now behaves like the rest of the program: it asks for the boundary if you have not selected one,
# then sits on an options prompt — `specify pad elevation, or [Surface]` — until you give it a number.
# The one-line form still works, because a transcript and a script want it.
#
# The surface is remembered between runs, so the common case (one ground surface, several pads at
# different levels) is just a number each time.

NEW

# --- a small site with real relief, built from definition points ---------------------------------
CMD SURFACECREATE Existing Ground
CMD SURFACEADDPOINT Existing Ground, 0, 0, 100
CMD SURFACEADDPOINT Existing Ground, 100, 0, 120
CMD SURFACEADDPOINT Existing Ground, 100, 100, 120
CMD SURFACEADDPOINT Existing Ground, 0, 100, 100
CMD SURFACEADDPOINT Existing Ground, 50, 50, 114
CMD SURFACEREBUILD Existing Ground
EXPECT SURFACES 1

CMD POLYLINE
CMD 20,20
CMD 80,20
CMD 80,80
CMD 20,80
CMD CLOSE
ESC
EXPECT POLYLINES 1
CHECK ALL

# --- bare PADSOLID with nothing selected ASKS, and stays running -----------------------------------
CMD PADSOLID
EXPECT LOG "select the closed polyline that bounds the pad"
ESC

# --- with the ring selected it goes straight to the options prompt ---------------------------------
BOX -500 -500 500 500
CMD PADSOLID
EXPECT LOG "specify pad elevation, or [Surface]"

# S names the surface and comes back to the same prompt, rather than ending the command.
CMD S
EXPECT LOG "enter surface name"
CMD Existing Ground
EXPECT LOG "specify pad elevation, or [Surface]"

# A word that is not a number and not an option keeps the prompt rather than throwing the command away.
CMD banana
EXPECT LOG "enter a pad elevation, or S to choose the surface"

# A number builds it. The pad crosses the ground, so there is a cut AND a fill.
CMD 110
EXPECT LOG "cut solid created"
EXPECT LOG "fill solid created"
EXPECT SOLIDS 2
CHECK ALL

# One undo takes both away: one command, one step.
CMD UNDO
EXPECT SOLIDS 0
CHECK ALL

# --- the one-line form still works, for a script ---------------------------------------------------
BOX -500 -500 500 500
CMD PADSOLID Existing Ground, 110
EXPECT SOLIDS 2
CHECK ALL
CMD UNDO

# --- and the short alias ----------------------------------------------------------------------------
BOX -500 -500 500 500
CMD PAD Existing Ground, 105
EXPECT LOG "cut solid created"
CHECK ALL

# --- refusals stay clear -----------------------------------------------------------------------------
CMD UNDO
BOX -500 -500 500 500
CMD PADSOLID Nope, 110
EXPECT LOG "no surface named "Nope""
EXPECT SOLIDS 0
CHECK ALL
