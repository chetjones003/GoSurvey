# REQ-157 — join, break, offset, fillet feature lines (ADR-035 (i)).

NEW
CMD FEATURELINE A
CMD 0,0,10
CMD 50,0,11
CMD END
CMD FEATURELINE B
CMD 50,0,11
CMD 100,0,12
CMD END
CMD JOINFEATURELINES 1 2
EXPECT LOG "JOINFEATURELINES — joined into one feature line"
CMD FEATURELINELIST
EXPECT LOG "3 vertices"

NEW
CMD FEATURELINE Split
CMD 0,0,10
CMD 50,0,11
CMD 100,0,12
CMD END
CMD BREAKFEATURELINE 1 2
EXPECT LOG "BREAKFEATURELINE — split into two feature lines"
CMD FEATURELINELIST
EXPECT LOG "Feature line 1"
EXPECT LOG "Feature line 2"

NEW
CMD FEATURELINE Off
CMD 0,0,10
CMD 100,0,12
CMD END
CMD OFFSETFEATURELINE 1 10
EXPECT LOG "OFFSETFEATURELINE — created offset copy"
CMD FEATURELINELIST
EXPECT LOG "Feature line 2"

NEW
CMD FEATURELINE Corner
CMD 0,0,10
CMD 50,0,11
CMD 50,50,12
CMD END
CMD FILLETFEATURELINE 1 2 10
EXPECT LOG "FILLETFEATURELINE — corner filleted"
CMD FLINSERTPI 1 25 0
EXPECT LOG "FLINSERTPI"
CMD FLHIGHLOW 1
EXPECT LOG "FLHIGHLOW"

NEW
CMD FEATURELINE ArcSeg
CMD 0,0,10
CMD 20,0,10
CMD END
CMD FLARC 1 1
EXPECT LOG "FLARC — converted to a 90-degree arc"
CMD OFFSETFEATURELINE 1 5
EXPECT LOG "OFFSETFEATURELINE — created offset copy"
CHECK ALL
