# REQ-154 — create feature lines from lines / arcs / polylines.

NEW
CMD LINE
CMD 0,0
CMD 100,0
ESC
BOX -1 -1 101 1
EXPECT SELECTED 1
CMD FEATURELINESFROMOBJECTS Toe KEEP
EXPECT LOG "FEATURELINESFROMOBJECTS — created 1 feature line"
CMD FEATURELINELIST
EXPECT LOG "Feature line 1: "Toe""
EXPECT LINES 1
CMD FLPROPERTIES 1
EXPECT LOG "FLPROPERTIES"

NEW
CMD LINE
CMD 0,0
CMD 50,0
ESC
BOX -1 -1 51 1
CMD FLFROM Ditch ERASE
EXPECT LOG "created 1 feature line"
EXPECT LINES 0
CMD FEATURELINELIST
EXPECT LOG "Feature line 1: "Ditch""

NEW
CMD ARC
PICK 0 0
PICK 10 10
PICK 20 0
ESC
BOX -1 -1 21 11
CMD FLFROM ArcFL KEEP
EXPECT LOG "created 1 feature line"
CMD FLPROPERTIES ArcFL
EXPECT LOG "FLPROPERTIES"
CHECK ALL
