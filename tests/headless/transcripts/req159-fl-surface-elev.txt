# REQ-159 — drape feature-line elevations from a TIN; relative offsets.

OPEN samples/surface-demo.gs
CMD SURFACECREATE EG, Ground + Curb
CMD SURFACEREBUILD EG
CMD FEATURELINE Ridge
CMD 30,40,0
CMD 70,40,0
CMD END
CMD FLELEV 1 FROMSURF EG
EXPECT LOG "FLELEV FROMSURF — applied surface elevations"
CMD FLELEV 1 RELATIVE EG
EXPECT LOG "FLELEV RELATIVE — offsets stored"
CMD FLPROPERTIES 1
EXPECT LOG "relative=EG"
SAVEAS %OUT%/req159-rel.gs
OPEN %OUT%/req159-rel.gs
CMD FLPROPERTIES 1
EXPECT LOG "relative=EG"
CMD FEATURELINE Other
CMD 0,0,5
CMD 10,0,5
CMD END
CMD FLELEV Other REF 1 1 1
EXPECT LOG "FLELEV REF"
CMD FLELEV 1 FROMSURF EG INSERT
EXPECT LOG "FLELEV FROMSURF"
CHECK ALL
