# GitHub issue #473 — BLOCKIMPORT of a standalone ACIS .sat file (Civil 3D ACISOUT), then INSERT.
#
# samples/CJ_4in_WELD_NECK_FLANGE.sat is a real 4" weld-neck flange: 16 analytic faces, two of
# them planar faces pierced by an 8-bolt circle. The .sat carries the model's absolute position
# from the drawing it was exported from (~4999 units); the importer re-centres it on the origin so
# INSERT places it at the cursor, not thousands of units away.

NEW
EXPECT SOLIDS 0

# BLOCKIMPORT defines the block; it does not place anything.
DIALOG OPEN %DATA%/CJ_4in_WELD_NECK_FLANGE.sat
CMD BLOCKIMPORT
EXPECT LOG "BLOCKIMPORT — imported"
EXPECT LOG CJ_4in_WELD_NECK_FLANGE
EXPECT SOLIDS 0
CMD BLOCKLIST
EXPECT LOG "CJ_4in_WELD_NECK_FLANGE"

# INSERT places the flange's solid at the given point — a solids-only block leaves no reference.
# The .sat is authored in inches and this drawing's INSUNITS default is feet, so INSERT scales the
# 0.75-unit-wide flange by 1/12; what matters here is that it lands AT (100, 200), not at ~4999.
CMD INSERT CJ_4in_WELD_NECK_FLANGE, 100, 200
EXPECT LOG "INSERT — placed"
EXPECT SOLIDS 1
EXPECT SOLIDKIND 0 Solid
EXPECT SOLIDBOUNDS 0 99.96875 199.96875 -0.0104 100.03125 200.03125 0.0104

# A second INSERT is a second flange.
CMD INSERT CJ_4in_WELD_NECK_FLANGE, 100, 210
EXPECT SOLIDS 2

UNDO
EXPECT SOLIDS 1
