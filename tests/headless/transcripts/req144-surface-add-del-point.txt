# REQ-144 — add and delete TIN definition points.

NEW

CMD SURFACECREATE Pts
CMD SURFACEADDPOINT Missing, 0, 0, 0
EXPECT LOG "no surface"
CMD SURFACEADDPOINT Pts, 0, 0, 0
CMD SURFACEADDPOINT Pts, 10, 0, 0
CMD SURFACEADDPOINT Pts, 10, 10, 5
CMD SURFACEADDPOINT Pts, 0, 10, 0
CMD SURFACEREBUILD Pts
EXPECT LOG "4 points"
CMD SURFACEDELPOINT Pts, 0, 0
CMD SURFACEREBUILD Pts
EXPECT LOG "3 points"
SAVEAS %OUT%/req144-pts.gs
NEW
OPEN %OUT%/req144-pts.gs
CMD SURFACEREBUILD Pts
EXPECT LOG "3 points"
CMD SURFACECREATE EmptyDel
CMD SURFACEDELPOINT EmptyDel
EXPECT LOG "has no definition points"
CMD SURFACECREATEGRID G1, 0, 0, 5, 5, 2, 2, 0, 0, 10, 10
CMD SURFACEADDPOINT G1, 0, 0, 0
EXPECT LOG "is not a TIN"
CMD SURFACEDELPOINT EmptyNone
EXPECT LOG "no surface"
CHECK ALL
