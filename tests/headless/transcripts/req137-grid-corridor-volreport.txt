NEW

OPEN samples/surface-demo.gs
CMD SURFACECREATECORR EmptyCorr
EXPECT LOG "no feature lines"
CMD SURFACELIST
EXPECT LOG "corridor"
CMD SURFACECREATEGRID G1, 0, 0, 5, 5, 2, 2, 0, 0, 10, 10
EXPECT LOG "grid"
CMD SURFACELIST
EXPECT LOG "grid"
CMD VOLREPORT
EXPECT LOG "no volume result"
CMD WATERDROP EXTRACT FL
EXPECT LOG "nothing to extract"
CMD SURFSWAPEDGE G1, 2.5, 2.5
EXPECT LOG "swapped an interior edge"
CMD SURFSWAPEDGE G1, 1000, 1000
EXPECT LOG "not on an interior edge"
